    /// <summary>
    /// Enumera os processadores logicos. A contagem vem das subchaves de
    /// HKLM\HARDWARE\DESCRIPTION\System\CentralProcessor (uma por processador logico), com
    /// fallback para Environment.ProcessorCount. Para 8 processadores logicos o Windows usa um
    /// unico grupo de processadores (grupos so sao criados acima de 64 LPs) â€” premissa
    /// documentada na interface. Em ambiente de desenvolvimento nao-Windows, um conjunto
    /// sintetico e criado para permitir revisao da interface (rotulado como tal).
    /// </summary>
    private void DetectTopology()
    {
        try
        {
            if (!OperatingSystem.IsWindows())
            {
                for (var i = 0; i < Math.Max(1, Environment.ProcessorCount); i++)
                {
                    _logicalProcessors.Add((0, i));
                }

                _cpuGateMessage =
                    "Ambiente nÃ£o-Windows (revisÃ£o do cÃ³digo): leituras indisponÃ­veis; interface em modo de revisÃ£o.";
                return;
            }

            using var baseKey = Registry.LocalMachine.OpenSubKey(@"HARDWARE\DESCRIPTION\System\CentralProcessor");
            var lpCount = baseKey?.SubKeyCount ?? Environment.ProcessorCount;
            for (var i = 0; i < lpCount; i++)
            {
                _logicalProcessors.Add((0, i)); // grupo 0: premissa documentada para <= 64 LPs
            }
        }
        catch (Exception ex)
        {
            var count = Math.Max(1, Environment.ProcessorCount);
            for (var i = 0; i < count; i++)
            {
                _logicalProcessors.Add((0, i));
            }

            _cpuGateMessage = $"Topologia nÃ£o pÃ´de ser lida ({ex.Message}); usando {count} processadores lÃ³gicos como premissa.";
        }
    }

    /// <summary>Preenche a aba de identificacao com dados do registro do Windows (somente leitura).</summary>
    private void FillIdentificationTab()
    {
        try
        {
            if (!OperatingSystem.IsWindows())
            {
                SetId("cpu", "NÃ£o disponÃ­vel â€” ambiente nÃ£o-Windows (revisÃ£o do cÃ³digo)");
                SetId("signature", "NÃ£o disponÃ­vel â€” " + UiTexts.CpuIdNote);

                SetId("os", "NÃ£o disponÃ­vel â€” ambiente nÃ£o-Windows");
                SetId("admin", _isAdmin ? "Sim" : "NÃ£o");
                return;
            }

            using var cpuKey = Registry.LocalMachine.OpenSubKey(@"HARDWARE\DESCRIPTION\System\CentralProcessor\0");
            var identifier = cpuKey?.GetValue("Identifier") as string ?? string.Empty;
            var brand = cpuKey?.GetValue("ProcessorNameString") as string ?? string.Empty;
            var mhz = cpuKey?.GetValue("~MHz") as int?;

            if (string.IsNullOrWhiteSpace(identifier))
            {
                SetId("cpu", "NÃ£o disponÃ­vel â€” chave de registro do processador ausente");
            }
            else
            {
                try
                {
                    _cpu = CpuSignature.Parse(identifier, string.IsNullOrWhiteSpace(brand) ? identifier : brand);
                    _cpuGateReason = _cpu.UnsupportedReason;
                    SetId("cpu", _cpu.BrandName);
                    SetId("signature", _cpu.Identifier);
                    SetId("fms", $"FamÃ­lia {_cpu.Family} (0x{_cpu.Family:X}) Â· Modelo {_cpu.Model} (0x{_cpu.Model:X}) Â· Stepping {_cpu.Stepping}"
                               + $" â€” alvo do projeto: FamÃ­lia 6, Modelo 0x8C (Tiger Lake-U i7-1195G7)");
                }
                catch (FormatException fex)
                {
                    _cpuGateReason = fex.Message;
                    SetId("signature", "NÃ£o disponÃ­vel â€” " + fex.Message);
                }

                if (mhz is not null)
                {
                    SetId("fms", GetId("fms") + $" Â· ~{mhz} MHz (informe do sistema)");
                }
            }

            SetId("arch", "x86-64 (Intel64) â€” " + System.Runtime.InteropServices.RuntimeInformation.ProcessArchitecture);
            SetId("lps", _logicalProcessors.Count + " processadores lÃ³gicos (grupo 0 â€” premissa documentada para <= 64 LPs)");
            SetId("cores", QueryCoreCount());

            // Sistema operacional: valores diretos do registro (ProductName e legado "Windows 10"
            // mesmo no Windows 11, por isso preferimos EditionID + DisplayVersion + Build).
            using var osKey = Registry.LocalMachine.OpenSubKey(@"SOFTWARE\Microsoft\Windows NT\CurrentVersion");
            var edition = osKey?.GetValue("EditionID") as string ?? "N/D";
            var display = osKey?.GetValue("DisplayVersion") as string ?? "N/D";
            var build = osKey?.GetValue("CurrentBuildNumber") as string ?? "N/D";
            var ubr = osKey?.GetValue("UBR") as int?;
            SetId("os", $"Windows ({edition}), versÃ£o {display}, build {build}.{(ubr?.ToString() ?? "N/D")} "
                      + $"â€” {System.Runtime.InteropServices.RuntimeInformation.OSDescription}");

            SetId("admin", _isAdmin
                ? "Sim (o aplicativo exige elevaÃ§Ã£o para o mecanismo de leitura da Fase 2)"
                : "NÃ£o");
        }
        catch (Exception ex)
        {
            SetId("cpu", "NÃ£o disponÃ­vel â€” falha na identificaÃ§Ã£o: " + ex.Message);
        }
    }

    /// <summary>Numero de nucleos fisicos via WMI (Win32_Processor). Falha = Nao disponivel com motivo.</summary>
    private string QueryCoreCount()
    {
        try
        {
            if (!OperatingSystem.IsWindows())
            {
                return "NÃ£o disponÃ­vel â€” ambiente nÃ£o-Windows";
            }

            using var searcher = new System.Management.ManagementObjectSearcher(
                "SELECT NumberOfCores, NumberOfLogicalProcessors FROM Win32_Processor");
            foreach (var item in searcher.Get())
            {
                using (item)
                {
                    var cores = item["NumberOfCores"]?.ToString();
                    var lps = item["NumberOfLogicalProcessors"]?.ToString();
                    if (cores is not null)
                    {
                        return $"{cores} nÃºcleos fÃ­sicos / {lps ?? "N/D"} processadores lÃ³gicos (Win32_Processor)";
                    }
                }
            }

            return "NÃ£o disponÃ­vel â€” WMI nÃ£o retornou dados";
        }
        catch (Exception ex)
        {
            return "NÃ£o disponÃ­vel â€” falha no WMI: " + ex.Message;
        }
    }

    private static bool TryParseHex(string? text, out ulong value)
    {
        value = 0;
        if (string.IsNullOrWhiteSpace(text))
        {
            return false;
        }

        var clean = text.Trim();
        if (clean.StartsWith("0x", StringComparison.OrdinalIgnoreCase))
        {
            clean = clean[2..];
        }

        clean = clean.Replace(" ", string.Empty).Replace("_", string.Empty);
        return clean.Length is >= 1 and <= 16
            && ulong.TryParse(clean, System.Globalization.NumberStyles.HexNumber, System.Globalization.CultureInfo.InvariantCulture, out value);
    }

    private async Task RunSamplingAsync()
    {
        if (_samplingBusy)
        {
            return;
        }

        if (_cpuGateReason is not null)
        {
            _lblLastUpdate.Text = "Leituras recusadas: " + _cpuGateReason;
            MessageBox.Show(this,
                "Leituras recusadas â€” hardware nÃ£o compatÃ­vel com o alvo do projeto." + Environment.NewLine
                + Environment.NewLine + _cpuGateReason,
                "IdentificaÃ§Ã£o do hardware", MessageBoxButtons.OK, MessageBoxIcon.Warning);
            return;
        }

        _samplingBusy = true;
        try
        {
            var snapshot = await Task.Run(_engine.CaptureSample);
            UpdateFromSample(snapshot);
        }
        catch (Exception ex)
        {
            _lblLastUpdate.Text = "Falha na amostragem: " + ex.Message;
        }
        finally
        {
            _samplingBusy = false;
        }
    }

using System.Text;

namespace IrisXe.Thermal.Core.Decoding;

/// <summary>
/// Utilitarios de manipulacao de bits para valores brutos de MSR de 64 bits.
/// Nenhuma interpretacao aqui e presumida: toda mascara usada pelos decodificadores
/// cita a documentacao Intel aplicavel ao Tiger Lake.
/// </summary>
public static class Bits
{
    /// <summary>Extrai <paramref name="width"/> bits a partir de <paramref name="loBit"/>.</summary>
    public static ulong Extract(ulong value, int loBit, int width)
        => width >= 64 ? value : (value >> loBit) & ((1UL << width) - 1UL);

    public static string Hex16(ulong value) => $"0x{value:X16}";

    /// <summary>Representacao binaria de 64 bits em grupos de 8, do bit 63 ao bit 0.</summary>
    public static string Bin64(ulong value)
    {
        var sb = new StringBuilder(71);
        for (var bit = 63; bit >= 0; bit--)
        {
            sb.Append(((value >> bit) & 1UL) == 1UL ? '1' : '0');
            if (bit > 0 && bit % 8 == 0)
            {
                sb.Append(' ');
            }
        }

        return sb.ToString();
    }

    public static string Range(int hi, int lo) => hi == lo ? $"[{hi}]" : $"[{hi}:{lo}]";
}

            + "externa) NAO pode ser confirmada por esta leitura.";

        string detail;

    private readonly List<(int Group, int Number)> _logicalProcessors;
    private readonly object _lock = new();

    public DiagnosticEngine(IMsrReader reader, AuditLog audit, IEnumerable<(int Group, int Number)> logicalProcessors)
    {
        _reader = reader;
        _audit = audit;
        _logicalProcessors = logicalProcessors.OrderBy(p => p.Group).ThenBy(p => p.Number).ToList();
    }

    public IMsrReader Reader => _reader;

    public AuditLog Audit => _audit;

    public SampleSnapshot? LastSample { get; private set; }

    /// <summary>Transicoes acumuladas desde o inicio da sessao â€” nunca limpas.</summary>
    public List<TransitionEvent> TransitionLog { get; } = new();

    public IReadOnlyList<(int Group, int Number)> LogicalProcessors => _logicalProcessors;

    /// Substitui o mecanismo de leitura (ex.: trocar de "Indisponivel" para "valores manuais").
    /// Somente leitura: qualquer leitor respeita o contrato IMsrReader, que nao possui escrita.
    /// </summary>
    public void SetReader(IMsrReader reader) =>
        _reader = reader ?? throw new ArgumentNullException(nameof(reader));

    /// <summary>
    /// Executa UMA rodada de amostragem (somente RDMSR conceitual; na Fase 1 os valores vem
    /// do mecanismo configurado â€” indisponivel ou manual/simulado). Registra auditoria,
    /// decodifica e detecta transicoes em relacao a amostra anterior.
    /// </summary>
    public SampleSnapshot CaptureSample()
    {
        lock (_lock)
        {
            var notes = new List<string>();
            var timestamp = DateTime.UtcNow;

            // --- 0x19C por processador logico (escopo documentado: por LP) ----------------
            var coreReadings = new List<LogicalProcessorReading>();
            int? tjMax = LastSample?.TemperatureTarget?.TjMaxC;
            foreach (var (group, number) in _logicalProcessors)
            {
                var result = _reader.Read(group, number, MsrAddress.ThermStatus);
                _audit.Add(new ReadingRecord
                {
                    TimestampUtc = result.TimestampUtc == default ? timestamp : result.TimestampUtc,
                    ProcessorGroup = group,
                    ProcessorNumber = number,
                    MsrAddress = MsrAddress.ThermStatus,
                    MsrName = MsrAddress.Name(MsrAddress.ThermStatus),
                    Mechanism = _reader.MechanismName,
                    Status = result.Success ? "Sucesso" : "Falha",
                    RawHex = result.Success ? result.ValueHex : null,
                    Error = result.ErrorMessage,
                    Win32Error = result.Win32Error,
                    Simulated = result.Simulated,
                });

                ThermStatusDecoding? decoded = null;
                if (result.Success)
                {
                    decoded = ThermStatusDecoding.Decode(result.Value, tjMax);
                }

                coreReadings.Add(new LogicalProcessorReading
                {
                    ProcessorGroup = group, ProcessorNumber = number, Result = result, Decoded = decoded,
                });
            }

            // --- 0x1B1 por LP para conferencia (escopo: pacote) ---------------------------
            var packageReadings = new List<LogicalProcessorReading>();
            MsrReadResult? firstPackageResult = null;
            foreach (var (group, number) in _logicalProcessors)
            {
                var result = _reader.Read(group, number, MsrAddress.PackageThermStatus);
                _audit.Add(new ReadingRecord
                {
                    TimestampUtc = result.TimestampUtc == default ? timestamp : result.TimestampUtc,
                    ProcessorGroup = group,
                    ProcessorNumber = number,
                    MsrAddress = MsrAddress.PackageThermStatus,
                    MsrName = MsrAddress.Name(MsrAddress.PackageThermStatus),
                    Mechanism = _reader.MechanismName,
                    Status = result.Success ? "Sucesso" : "Falha",
                    RawHex = result.Success ? result.ValueHex : null,
                    Error = result.ErrorMessage,
                    Win32Error = result.Win32Error,
                    Simulated = result.Simulated,
                });

                if (result.Success && firstPackageResult is null)
                {
                    firstPackageResult = result;
                }

                packageReadings.Add(new LogicalProcessorReading
                {
                    ProcessorGroup = group,
                    ProcessorNumber = number,
                    Result = result,
                    Decoded = null, // pacote nao tem decodificacao por-LP; exibida uma vez, abaixo
                });
            }

            var packageDecoded = firstPackageResult is not null
                ? PackageThermStatusDecoding.Decode(firstPackageResult.Value, tjMax)
                : null;

            // Conferencia: leituras do pacote divergentes entre LPs NAO sao descartadas.
            var distinctPackage = packageReadings
                .Where(r => r.Result.Success)
                .Select(r => r.Result.Value)
                .Distinct()
                .ToList();
            if (distinctPackage.Count > 1)
            {
                notes.Add(
                    "MSR_PACKAGE_THERM_STATUS (0x1B1) retornou valores diferentes entre processadores logicos "
                    + $"({string.Join(", ", distinctPackage.Select(Bits.Hex16))}). Escopo documentado e por pacote; "
                    + "a divergencia NAO foi descartada e deve ser investigada (pode indicar leitura em momentos "
                    + "diferentes do mesmo contador ou comportamento de virtualizacao).");
            }

            // --- 0x1A2 por LP para conferencia (valores de fabrica devem ser iguais) ------
            var targetReadings = new List<MsrReadResult>();
            TemperatureTargetDecoding? targetDecoded = null;
            foreach (var (group, number) in _logicalProcessors)
            {
                var result = _reader.Read(group, number, MsrAddress.TemperatureTarget);
                _audit.Add(new ReadingRecord
                {
                    TimestampUtc = result.TimestampUtc == default ? timestamp : result.TimestampUtc,
                    ProcessorGroup = group,
                    ProcessorNumber = number,
                    MsrAddress = MsrAddress.TemperatureTarget,
                    MsrName = MsrAddress.Name(MsrAddress.TemperatureTarget),
                    Mechanism = _reader.MechanismName,
                    Status = result.Success ? "Sucesso" : "Falha",
                    RawHex = result.Success ? result.ValueHex : null,
                    Error = result.ErrorMessage,
                    Win32Error = result.Win32Error,
                    Simulated = result.Simulated,
                });

                targetReadings.Add(result);
                if (result.Success && targetDecoded is null)
                {
                    targetDecoded = TemperatureTargetDecoding.Decode(result.Value);
                }
            }

            var distinctTargets = targetReadings
                .Where(r => r.Success)
                .Select(r => r.Value)
                .Distinct()
                .ToList();
            if (distinctTargets.Count > 1)
            {
                notes.Add(
                    "MSR_TEMPERATURE_TARGET (0x1A2) retornou valores diferentes entre processadores logicos "
                    + $"({string.Join(", ", distinctTargets.Select(Bits.Hex16))}). Esperado: valor calibrado em fabrica "
                    + "identico. A divergencia NAO foi descartada â€” cada valor permanece no log de auditoria.");
            }

            if (coreReadings.All(r => !r.Result.Success))
            {
                notes.Add(
                    "Nenhuma leitura de IA32_THERM_STATUS foi bem-sucedida nesta amostragem. Os campos exibirao "
                    + "'Nao disponivel' com o motivo registrado no log de auditoria (nenhum valor presumido e exibido).");
            }

            var snapshot = new SampleSnapshot
            {
                TimestampUtc = timestamp,
                CoreReadings = coreReadings,
                PackageReadings = packageReadings,
                TemperatureTargetReadings = targetReadings,
                TemperatureTarget = targetDecoded,
                Package = packageDecoded,
                Notes = notes,
                MechanismName = _reader.MechanismName,
                Simulated = _reader.IsSimulated,
            };

            // Transicoes em relacao a amostra anterior (requisito Â§6) â€” nunca limpas.
            foreach (var transition in TransitionDetector.Diff(LastSample, snapshot))
            {
                TransitionLog.Add(transition);
            }

            LastSample = snapshot;
            return snapshot;
        }
    }

public static class ThrottleStopComparison
{
    /// <summary>Extrai uma linha de base rotulada de uma amostra.</summary>
    public static ThrottleStopBaseline FromSample(SampleSnapshot sample, string label)
    {
        var targetByLp = new Dictionary<int, ulong>();
        for (var i = 0; i < sample.TemperatureTargetReadings.Count; i++)
        {
            var reading = sample.TemperatureTargetReadings[i];
            var lp = i < sample.CoreReadings.Count ? sample.CoreReadings[i].ProcessorNumber : i;
            if (reading.Success)
            {
                targetByLp[lp] = reading.Value;
            }
        }

        var thermByLp = new Dictionary<int, ulong>();
        foreach (var reading in sample.CoreReadings)
        {
            if (reading.Result.Success)
            {
                thermByLp[reading.ProcessorNumber] = reading.Result.Value;
            }
        }

        return new ThrottleStopBaseline
        {
            Label = label,
            TimestampUtc = sample.TimestampUtc,
            TemperatureTargetByLp = targetByLp,
            ThermStatusByLp = thermByLp,
            PackageThermStatus = sample.PackageReadings.FirstOrDefault(r => r.Result.Success) is { } pkgReading
                ? pkgReading.Result.Value
                : null,
            Simulated = sample.Simulated,
        };
    }

    public static ComparisonReport Compare(ThrottleStopBaseline before, ThrottleStopBaseline after)
    {
        var entries = new List<ComparisonEntry>();

        foreach (var lp in before.TemperatureTargetByLp.Keys.Union(after.TemperatureTargetByLp.Keys).OrderBy(k => k))
        {
            var hasBefore = before.TemperatureTargetByLp.TryGetValue(lp, out var beforeValue);
            var hasAfter = after.TemperatureTargetByLp.TryGetValue(lp, out var afterValue);
            entries.Add(new ComparisonEntry
            {
                MsrAddress = MsrAddress.TemperatureTarget,
                MsrName = MsrAddress.Name(MsrAddress.TemperatureTarget),
                ProcessorNumber = lp,
                BeforeHex = hasBefore ? Bits.Hex16(beforeValue) : "N/D",
                AfterHex = hasAfter ? Bits.Hex16(afterValue) : "N/D",
                Changed = hasBefore && hasAfter && beforeValue != afterValue,
            });
        }

        foreach (var lp in before.ThermStatusByLp.Keys.Union(after.ThermStatusByLp.Keys).OrderBy(k => k))
        {
            var hasBefore = before.ThermStatusByLp.TryGetValue(lp, out var beforeValue);
            var hasAfter = after.ThermStatusByLp.TryGetValue(lp, out var afterValue);
            entries.Add(new ComparisonEntry
            {
                MsrAddress = MsrAddress.ThermStatus,
                MsrName = MsrAddress.Name(MsrAddress.ThermStatus),
                ProcessorNumber = lp,
                BeforeHex = hasBefore ? Bits.Hex16(beforeValue) : "N/D",
                AfterHex = hasAfter ? Bits.Hex16(afterValue) : "N/D",
                Changed = hasBefore && hasAfter && beforeValue != afterValue,
            });
        }

        if (before.PackageThermStatus is not null || after.PackageThermStatus is not null)
        {

                MsrAddress = MsrAddress.PackageThermStatus,
                MsrName = MsrAddress.Name(MsrAddress.PackageThermStatus),

                       && after.PackageThermStatus is not null
                       && before.PackageThermStatus != after.PackageThermStatus,
            });

        foreach (var reading in current.CoreReadings)

        foreach (var r in records)

/// <summary>
/// Assinatura do processador equivalente a CPUID.1:EAX (famÃ­lia/modelo/stepping).
///
/// Estrategia (Fase 1, sem driver): a assinatura e obtida da cadeia "Identifier" do registro
/// Windows (HKLM\HARDWARE\DESCRIPTION\System\CentralProcessor\0), que o proprio sistema
/// operacional deriva da instrucao CPUID durante o boot â€” ex.: "Intel64 Family 6 Model 140
/// Stepping 1". Isso equivale a consultar CPUID sem executar codigo nativo de usuario.
/// O nome comercial vem de "ProcessorNameString" na mesma chave.
///
/// Tiger Lake-U (i7-1195G7): Family 6, Model 0x8C (140 decimal).
/// Leituras em outros processadores sao RECUSADAS (requisito 9.1 â€” impedir leituras
/// incompativeis com o hardware detectado).
/// </summary>
public sealed record CpuSignature
{
    /// <summary>Model CPUID do Tiger Lake-U (i7-1195G7): 0x8C = 140.</summary>
    public const int TigerLakeUModel = 0x8C;

    public const int IntelFamily = 6;

    private static readonly Regex IdentifierRegex =
        new(@"Family\s+(?<family>\d+)\s+Model\s+(?<model>\d+)\s+Stepping\s+(?<stepping>\d+)",
            RegexOptions.IgnoreCase | RegexOptions.CultureInvariant | RegexOptions.Compiled);

    public required string Identifier { get; init; }

    public required string BrandName { get; init; }

    public required int Family { get; init; }

    public required int Model { get; init; }

    public required int Stepping { get; init; }

    public bool IsIntelFamily6 =>
        Identifier.StartsWith("Intel64", StringComparison.OrdinalIgnoreCase) && Family == IntelFamily;

    public bool IsTigerLake1195G7 => IsIntelFamily6 && Model == TigerLakeUModel;

    public string Description => $"{BrandName} â€” {Identifier} (FamÃ­lia {Family}, Modelo 0x{Model:X}, Stepping {Stepping})";

    public string? UnsupportedReason
    {
        get
        {
            if (!Identifier.StartsWith("Intel64", StringComparison.OrdinalIgnoreCase))
            {
                return $"Processador nÃ£o Intel64 ({Identifier}). As leituras foram recusadas: o hardware detectado nÃ£o Ã© o alvo do projeto.";
            }

            if (Family != IntelFamily)
            {
                return $"FamÃ­lia CPUID {Family} nÃ£o suportada (esperado: 6). As leituras foram recusadas.";
            }

            if (Model != TigerLakeUModel)
            {
                return $"Modelo CPUID 0x{Model:X} nÃ£o Ã© o Tiger Lake-U 0x8C do i7-1195G7. As leituras foram recusadas: "
                     + "a decodificaÃ§Ã£o foi validada apenas para este processador (requisito 9.1).";
            }

            return null;
        }
    }

    /// <summary>Analisador da cadeia "Identifier" do registro do Windows.</summary>
    /// <exception cref="FormatException">Quando a cadeia nao contem a assinatura familia/modelo/stepping.</exception>
    public static CpuSignature Parse(string identifier, string brandName)
    {
        var match = IdentifierRegex.Match(identifier);
        if (!match.Success)
        {
            throw new FormatException(
                $"NÃ£o foi possÃ­vel interpretar a assinatura do processador a partir de '{identifier}'. "
                + "Formato esperado: 'Intel64 Family 6 Model 140 Stepping 1'.");
        }

        return new CpuSignature
        {
            Identifier = identifier.Trim(),
            BrandName = brandName.Trim(),
            Family = int.Parse(match.Groups["family"].Value, CultureInfo.InvariantCulture),
            Model = int.Parse(match.Groups["model"].Value, CultureInfo.InvariantCulture),
            Stepping = int.Parse(match.Groups["stepping"].Value, CultureInfo.InvariantCulture),
        };
    }
